123template<class TA, class TB>
124wbMatrix<T>& wbMatrix<T>::init(
125 const char *F, int L,
126 const wbMatrix<TA> &M, const wbvector<TB> &V, char dim
127){
128 size_t i,j,n1=M.dim1,n2=M.dim2;
129 if (abs(dim)==2) {
130 if (n1!=V.len) wblog(F_L,
131 "ERR %s() size mismatch %d/%dx%d",FCT,V.len,n1,n2);
132 RENEW(n1,n2+1,0,0);
133 if (dim>0) {
134 for (i=0; i<n1; ++i) {
135 for (j=0; j<n2; ++j) (*this)(i,j)=(T)M(i,j);
136 (*this)(i,j)=(T)V[i];
137 }
138 }
139 else {
140 for (i=0; i<n1; ++i) {
141 (*this)(i,0)=(T)V[i];
142 for (j=0; j<n2; ++j) (*this)(i,j+1)=(T)M(i,j);
143 }
144 }
145 }
146 else if (abs(dim)==1) {
147 if (n2!=V.len) wblog(F_L,
148 "ERR %s() size mismatch %dx%d/%d",FCT,n1,n2,V.len);
149 RENEW(n1+1,n2,0,0);
150 if (dim>0) {
151 for (i=0; i<n1; ++i)
152 for (j=0; j<n2; ++j) (*this)(i+1,j)=(T)M(i,j);
153 for (j=0; j<n2; ++j) (*this)(i,j)=(T)V[j];
154 }
155 else {
156 for (j=0; j<n2; ++j) (*this)(0,j)=(T)V[j];
157 for (i=0; i<n1; ++i)
158 for (j=0; j<n2; ++j) (*this)(i+1,j)=(T)M(i,j);
159 }
160 }
161 else wblog(F_L,"ERR %s() invalid dim=%d",FCT,dim);
162
163 return *this;

221template<>
222WBINDEX& wbMatrix<size_t>::toIndex(
223 WBINDEX &I, const WBINDEX *S) const {
224
225 const size_t *idx=data;
226 if (S==NULL) {
227 WBINDEX SX(dim2); widx_t *s=SX.data; size_t i=0,j;
228
229 for (; i<dim1; ++i, idx+=dim2) {
230 for (j=0; j<dim2; ++j) {
231 if (s[j]<idx[j]) s[j]=idx[j];
232 }
233 }
234 SX+=1; return toIndex(I,&SX);
235 }
236
237 I.init(dim1); if (!dim1) return I;
238 if (!dim2) wblog(FL,
239 "ERR %s() got empty index matrix (%dx%d)",FCT,dim1,dim2);
240
241 size_t i=0,j, l=dim2-1;
242 const widx_t *s=S->data;
243
244 for (; i<dim1; ++i, idx+=dim2) { widx_t &q=I.data[i];
245 for (j=l, q=idx[j--]; j<l; --j) { q = q*size_t(s[j]) + size_t(idx[j]); }
246 }
247
248 return I;

251template<>
252WBIDXMAT& wbMatrix<size_t>::toIndex2D(
253 const WBINDEX &S, const wbindex &Ic,
254 WBIDXMAT &IJ, char pos
255 ) const {
256
257 wbvector<char> M(S.len);
258 size_t i; char *m=M.data;
259
260 if (S.len!=dim2) wblog(FL,
261 "ERR %s() size mismatch (%dx%d/%d)",FCT,dim1,dim2,S.len);
262 if (!dim2) wblog(FL,
263 "ERR %s() got empty index matrix (%dx%d)",FCT,dim1,dim2);
264
265 for (i=0; i<Ic.len; ++i) {
266 if (Ic[i]>=S.len) wblog(FL,"ERR %s() index out of bounds "
267 "(%s; %d)",FCT,STR(Ic+1),S.len);
268 if ((++m[Ic[i]])>1) wblog(FL,"ERR %s() index not unique "
269 "(%s; %d)",FCT,STR(Ic+1),S.len
270 );
271 }
272
273 if ((--pos)>1) wblog(FL,
274 "ERR %s() index out of founds (pos=%d)",FCT,pos);
275
276 if (dim2==2 && Ic.len==1) {
277 if (int(Ic[0])==pos) { IJ=(*this); }
278 else {
279 IJ.init(dim1,dim2);
280 size_t d12=dim1*dim2; widx_t *d=IJ.data;
281 for (i=0; i<d12; i+=2) {
282 d[i]=data[i+1];
283 d[i+1]=data[i];
284 }
285 }
286 return IJ;
287 }
288
289 IJ.init(dim1,2); if (!dim1) return IJ;
290
291 wbindex Ik(S.len-Ic.len);
292 unsigned j=0, k=0, lk=Ik.len-1, lc=Ic.len-1;
293 const size_t *idx=data, *s=S.data;
294 const widx_t *ic=Ic.data; widx_t *ik=Ik.data;
295
296 for (i=0; i<S.len; ++i) { if (!m[i]) { ik[j++]=i; }}
297
298 for (i=0; i<dim1; ++i, idx+=dim2, k+=2) {
299 if (Ik.len) {
300 widx_t &I=IJ.data[k+(1-pos)]; I=idx[ik[lk]];
301 for (j=lk-1; j<lk; --j) { if (I) I*=s[ik[j]]; I+=idx[ik[j]]; }
302 }
303 if (Ic.len) {
304 widx_t &J=IJ.data[k+pos]; J=idx[ic[lc]];
305 for (j=lc-1; j<lc; --j) { if (J) J*=s[ic[j]]; J+=idx[ic[j]]; }
306 }
307 }
308
309 return IJ;
310};

414

417

420

422 maxdiff=MAX(maxdiff, fabs(double(M.data[i]-data[i])));
423 maxval=MAX(maxval,

425 }
426
427 return ((maxval==0.) ? 0. : maxdiff/maxval);

623template <class T> inline
