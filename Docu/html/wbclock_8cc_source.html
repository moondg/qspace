28inline wbstring Wb::sec2Str(double t) {
29 wbstring st(16);
30 unsigned l;
31
32 if (t<100) { l=snprintf(st.data,st.len,"%.5g",t); } else
33 if (t<3600) {
34 double s=fmod(t,60), m=(t-s)/60;
35 if (::fabs(s-::round(s))<1e-6)
36 { l=snprintf(st.data,st.len,"%02.0f:%02f", m,s); }
37 else { l=snprintf(st.data,st.len,"%02.0f:%02.3f",m,s); }
38 }
39 else {
40 double s=fmod(t,60);
41 unsigned q=::round((t-s)/60), m,h,d;
42
43 m=q%60; q=(q-m)/60;
44 h=q%24; d=(q-h)/24;
45
46 if (d)
47 l=snprintf(st.data,st.len,"%d-%02d:%02d:%02.0f",d,h,m,s);
48 else l=snprintf(st.data,st.len, "%02d:%02d:%02.0f", h,m,s);
49 }
50
51 if (l>=st.len) wblog(FL,"WRN %s() string out of bounds "
52 "(len=%d/%d)\nstr = %s",FCT,l,st.len,st.data);
53 return st;
54};

114Wb::Clock* Wb::ClockSet::use(const char *istr, char mode) {
115 Wb::Clock* clk;
116
117 if (!istr ) wblog(FL,"ERR %s() got null clock string", FCT);
118 if (!istr[0]) wblog(FL,"WRN %s() got empty clock string",FCT);
119
120 auto i=buf.find(istr);
121 if (i!=buf.end()) { clk=i->second; }
122 else {
123 clk = new Wb::Clock(istr,mode&2,0,NULL,NULL,0);
124 auto q=buf.insert({ istr, clk });
125 if (!q.second) wblog(FL,"WRN %s() failed to insert clock",FCT);
126 }
127
128 if (mode>1) {
129 if (mode&1) { clk->resume(); }
130 if (mode>3) wblog(FL,"WRN %s() got mode = %s",FCT,cSTR(mode));
131 }
132 return clk;
