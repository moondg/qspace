225double Wb::Clock::gettime(char cflag) const {
226
227 double dt=0;
228
229 if (cflag) {
230 dt=ctot; if (ncall%2) {
231 dt+=double(clock()-cref)/CLOCKS_PER_SEC; }
232 }
233 else {
234 dt=ttot; if (ncall%2) {
235 dt+=get_dt_wall(); }
236 }
237

241void Wb::Clock::info(const char *istr, char vflag) const {
242
243 static int first_call=1;
244
245 if (vflag>8) {
246 if (vflag=='v') { vflag=1; } else
247 if (vflag=='V') { vflag=2; } else
248 wblog(FL,"WRN %s() unexpected vflag=%s",FCT,cSTR(vflag));
249 }
250 if (!ncall && vflag<2) { return; }
251
252 unsigned l=0, n=16; char s[n];
253 double tc=gettime('c'), tw=gettime();
254
255 if (nz) {
256 double p0=1-nz/(double)(ncall/2);
257 l=snprintf(s,16,"/%5.1f%%", 100*p0);
258 } else s[0]=0;
259
260 if (clk_flags && l<n) {
261 l+=snprintf(s+l,n-l,"/%d", clk_flags); }
262
263 if (first_call) { first_call=0;
264 PRINTF("\n %-28s Count CPU-time Wall-time / call\n",
265 "Wb::Clocks");
266 }
267
268 PRINTF(" %c %-26s%8ld %10s %10s %9.3g %s\n",
269 (ncall%2) ? '*':' ', istr? istr:name.data, ncall/2,
270 sec2Str(tc).data, sec2Str(tw).data,
271 tw/(ncall/2), s
272 );

275double Wb::Clock::getTickFreq() const {
276
277 unsigned i,j,m=512,n=32;
278 clock_t c1,c2,nc=0;
279
280 CHRONO_CLOCK t1,t2;
281 double dt=0;
282
283 for (i=j=0; i<n; ++i) {
284 t1=CHRONO_NOW(); c1=clock(); while (c1==clock()) { if (++j>m) break; };
285 t2=CHRONO_NOW(); c2=clock();
286
287 if (i) {
288 nc+=(c2-c1);
289 dt+=CHRONO_DIFF2SEC( t2-t1 );
290 }
291 }
292
293 return (dt/nc);

321 Profile(const char *F, int L, const char *istr=0) {
322 unsigned i=0, l=0, n=32; char fl[32];
323 if (!F || !F[0] || !L) wblog(FL,"ERR %s() invalid empty F !?",FCT);
324
325 for (; F[i]; ++i) { if (F[i]=='/') { l=i+1; }}
326 for (i=0; F[l] && i<n; ++l, ++i) {
327 if (isalpha(F[l])) { fl[i]=F[l]; } else
328 if ((F[l]=='.' || F[l]=='_') && i) { fl[i]='_'; } else { fl[i]='X'; }
329 }; l=i;
330
331 if (l<n) { l+=snprintf(fl+l,n-l,"_%03d",L); }
332 if (l<n && istr && istr[0]) { l+=snprintf(fl+l,n-l,":%s",istr); }
333 if (l>=n) { fl[n-1]=0; wblog(FL,
334 "ERR %s() string out of bounds (%d/%d)\n%s",FCT,l,n,fl);
335 }
336 std::string fl_=fl;
337
338 #pragma omp critical (gwb_Profs__)
339 { auto im=gwb_Profs.find(fl_);
340 if (im==gwb_Profs.end()) {
341 gwb_Profs[fl_]=map<int, Wb::line_stat__ >();
342 if ((im=gwb_Profs.find(fl_))==gwb_Profs.end())
343 { wblog(FL,"ERR %s() map entry failed !?",FCT); }
344 }
345 M=&(im->second);
346 }
347 tic();
348 };

354 #pragma omp critical (gwb_ProfL__)

357 if ((im=M->find(L))==M->end())
358 { wblog(FL,"ERR %s() map entry failed !?",FCT); }
359 }

361 }
